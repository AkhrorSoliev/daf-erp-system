# B4 «To'lovlar» Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Moliya → «To'lovlar» (`/payments/list`) from the approved mock-ups: every payment of a period in one list with period chips, filters, a «Tanlangan» total, a per-row «Nimaga ketdi» split from the one cash engine, a drawer with receipt numbers, balance before → after, correction and CEO-only cancellation with a required reason; retire «O'quvchi to'lovi» and «Kutilyotgan to'lovlar».

**Architecture:** The ledger replay inside `getIncomeMonthAttribution` moves to a pure `attributeIncome` that also returns parts per payment; the overview card and the new page call the same function, so the page's rows add up to «Kassaga tushdi». A new `PaymentsListController` (`/payments/list/*`) builds every row through one `buildRows` path (SQL filters → engine once → payment-day groups → in-memory filters → totals → page) for the list, the drawer, the summary card and the Excel. Receipt codes are given after commit by a `payment.received` listener; a correction is linked by a new `Payment.correctsPaymentId` column.

**Tech Stack:** NestJS + Prisma 7 + Jest + exceljs (server); Next.js 16 + React Query 5 + Vitest + shadcn (client). No new dependency.

**Code in this plan is written compactly; `npx prettier --write` (server only) reformats it. Never run prettier on `client/`.**

```ts
import { Prisma, TransactionType } from '@prisma/client';
import type { PrismaService } from '../../prisma/prisma.service';
import { tashkentMonthKey } from '../date/tashkent';
import type { ResolvedPeriod } from './period-helpers';
import { branchIdWhere, type ReportBranchIds } from './report-branch-scope';

export interface AttributionRow { studentId: number | null; type: TransactionType; amount: number; branchId: number | null; createdAt: Date; paymentId: string | null }
export interface PaymentParts { currentMonth: number; advance: number; late: { monthKey: string; amount: number }[] }
export interface IncomeAttribution {
  currentMonth: number; advance: number; advanceStudents: number; lateByMonth: Map<string, number>;
  paymentCount: number; latePaymentCount: number; lateStudentCount: number; byPayment: Map<string, PaymentParts>;
}

export async function loadAttributionRows(db: PrismaService, companyId: number, period: ResolvedPeriod, branchIds: ReportBranchIds) {
  const payers = await db.payment.groupBy({
    by: ['studentId'],
    where: { companyId, status: 'COMPLETED', createdAt: { gte: period.start, lte: period.endTs }, ...branchIdWhere(branchIds) },
  });
  const payerIds = payers.map((p) => p.studentId).filter((id): id is number => id != null);
  if (payerIds.length === 0) return { payerIds, rows: [] as AttributionRow[] };
  // Reversed originals AND their reversal rows are both excluded (corrected history).
  const rows: AttributionRow[] = await db.transaction.findMany({
    where: { companyId, studentId: { in: payerIds }, reversedAt: null, reversedTransactionId: null },
    select: { studentId: true, type: true, amount: true, branchId: true, createdAt: true, paymentId: true },
    orderBy: [{ createdAt: 'asc' }, { id: 'asc' }],
  });
  return { payerIds, rows };
}

export function attributeIncome(rows: readonly AttributionRow[], period: ResolvedPeriod, branchIds: ReportBranchIds): IncomeAttribution {
  const boundaryKey = tashkentMonthKey(period.start);
  const startMs = period.start.getTime();
  const endMs = period.endTs.getTime();
  const byStudent = new Map<number, AttributionRow[]>();
  for (const r of rows) {
    if (r.studentId == null) continue;
    const list = byStudent.get(r.studentId);
    if (list) list.push(r); else byStudent.set(r.studentId, [r]);
  }
  const lateByMonth = new Map<string, number>();
  const acc = new Map<string, { currentMonth: number; advance: number; late: Map<string, number> }>();
  const partsOf = (id: string) => {
    let p = acc.get(id);
    if (!p) { p = { currentMonth: 0, advance: 0, late: new Map() }; acc.set(id, p); }
    return p;
  };
  let currentMonth = 0, advance = 0, advanceStudents = 0, paymentCount = 0, latePaymentCount = 0;
  const lateStudents = new Set<number>();

  for (const [studentId, list] of byStudent) {
    const debts: { monthKey: string; remaining: number }[] = [];
    let debtHead = 0;
    const credits: { amount: number; inScope: boolean; paymentId: string | null }[] = [];
    let creditHead = 0;
    let studentAdvance = 0;
    for (const r of list) {
      if (r.amount === 0) continue; // LESSON_CONSUMPTION — no balance movement.
      const ts = r.createdAt.getTime();
      const inPeriod = ts >= startMs && ts <= endMs;
      if (r.amount < 0) {
        let debit = -r.amount;
        while (debit > 0 && creditHead < credits.length) {
          const frag = credits[creditHead];
          const taken = Math.min(debit, frag.amount);
          frag.amount -= taken;
          debit -= taken;
          if (frag.inScope && inPeriod) {
            studentAdvance -= taken;
            currentMonth += taken;
            if (frag.paymentId) { const p = partsOf(frag.paymentId); p.advance -= taken; p.currentMonth += taken; }
          }
          if (frag.amount === 0) creditHead++;
        }
        if (debit > 0) debts.push({ monthKey: tashkentMonthKey(r.createdAt), remaining: debit });
        continue;
      }
      let credit = r.amount;
      const attribute = r.type === TransactionType.PAYMENT && inPeriod &&
        (branchIds === null || (r.branchId != null && branchIds.includes(r.branchId)));
      const parts = attribute && r.paymentId ? partsOf(r.paymentId) : null;
      let settledOldDebt = false;
      while (credit > 0 && debtHead < debts.length) {
        const frag = debts[debtHead];
        const taken = Math.min(credit, frag.remaining);
        frag.remaining -= taken;
        credit -= taken;
        if (attribute) {
          if (frag.monthKey < boundaryKey) {
            lateByMonth.set(frag.monthKey, (lateByMonth.get(frag.monthKey) ?? 0) + taken);
            if (parts) parts.late.set(frag.monthKey, (parts.late.get(frag.monthKey) ?? 0) + taken);
            settledOldDebt = true;
          } else {
            currentMonth += taken;
            if (parts) parts.currentMonth += taken;
          }
        }
        if (frag.remaining === 0) debtHead++;
      }
      if (attribute) {
        paymentCount++;
        if (settledOldDebt) { latePaymentCount++; lateStudents.add(studentId); }
      }
      if (credit > 0) {
        if (attribute) { studentAdvance += credit; if (parts) parts.advance += credit; }
        credits.push({ amount: credit, inScope: attribute, paymentId: attribute ? r.paymentId : null });
      }
    }
    advance += studentAdvance;
    if (studentAdvance > 0) advanceStudents++;
  }
  const byPayment = new Map<string, PaymentParts>();
  for (const [id, p] of acc) {
    byPayment.set(id, {
      currentMonth: p.currentMonth, advance: p.advance,
      late: [...p.late].map(([monthKey, amount]) => ({ monthKey, amount })).sort((a, b) => (a.monthKey < b.monthKey ? 1 : -1)),
    });
  }
  return { currentMonth, advance, advanceStudents, lateByMonth, paymentCount, latePaymentCount, lateStudentCount: lateStudents.size, byPayment };
}
```

import { addDaysToDateStr, tashkentDateStr, tashkentDayStartUtc } from '../../common/date/tashkent';
import { enrollmentStatusOn, EnrollmentStatusEvent, supplyClosingRow, supplyOpeningRow } from '../../students/shared/enrollment-status-on';

/**
 * «Guruh» of a payment = the groups the student was in ON THE PAYMENT'S
 * TASHKENT DAY (spec B4 §4.5): an enrollment counts if it was ACTIVE at any
 * moment of that day. The log's two ends are filled like the departures loader
 * (`reports/shared/departures.loader.ts`), and a deleted group closes the
 * enrollment at `group.deletedAt`.
 */
export interface DayEnrollment { id: string; groupId: string; status: string; createdAt: Date; startDate: Date | null; statusChangedAt: Date | null; groupDeletedAt: Date | null; log: EnrollmentStatusEvent[] }

export function activeOnDay(e: DayEnrollment, day: string): boolean {
  const dayStart = tashkentDayStartUtc(day);
  const dayEnd = tashkentDayStartUtc(addDaysToDateStr(day, 1));
  if (e.startDate && e.startDate.getTime() >= dayEnd.getTime()) return false;
  if (e.groupDeletedAt && e.groupDeletedAt.getTime() < dayStart.getTime()) return false;
  const log = e.log.length ? e.log.map((x) => ({ ...x })) : [{ status: 'ACTIVE', transitionAt: e.createdAt }];
  supplyOpeningRow(log, e.createdAt);
  supplyClosingRow(log, e);
  if (enrollmentStatusOn(log, dayStart, e) === 'ACTIVE') return true;
  return log.some((x) => x.status === 'ACTIVE' && x.transitionAt >= dayStart && x.transitionAt < dayEnd);
}

export function groupIdsOnDay(enrollments: readonly DayEnrollment[], paidAt: Date): string[] {
  const day = tashkentDateStr(paidAt);
  return [...new Set(enrollments.filter((e) => activeOnDay(e, day)).map((e) => e.groupId))];
}

```ts
import { Prisma } from '@prisma/client';
import type { PrismaService } from '../prisma/prisma.service';
import { tashkentMonthKey } from '../common/date/tashkent';
import { isTransactionConflict } from '../common/transaction-conflict';

/**
 * Receipt numbers (spec B4 §4.7). `receiptCode` is unique across the whole
 * table, so the next number is the GLOBAL highest of the year's prefix + 1,
 * and the year is the Tashkent year (Railway runs in UTC). Allocated in its own
 * Serializable transaction, never inside the money transaction: a numbering
 * clash must not refuse a payment. Retried on a clash.
 */
const TRIES = 3;
const TX = { isolationLevel: Prisma.TransactionIsolationLevel.Serializable, maxWait: 5_000, timeout: 10_000 };
const retryable = (e: unknown) => isTransactionConflict(e) || (e instanceof Prisma.PrismaClientKnownRequestError && e.code === 'P2002');

export function nextSequence(lastCode: string | null, prefix: string): string {
  if (!lastCode) return '00001';
  const n = parseInt(lastCode.replace(prefix, ''), 10);
  return Number.isNaN(n) ? '00001' : String(n + 1).padStart(5, '0');
}

async function withRetry<T>(run: () => Promise<T>): Promise<T> {
  for (let attempt = 1; ; attempt++) {
    try { return await run(); } catch (e) { if (attempt >= TRIES || !retryable(e)) throw e; }
  }
}

export function allocatePaymentReceiptCode(prisma: PrismaService, paymentId: string): Promise<string | null> {
  return withRetry(() => prisma.$transaction(async (tx) => {
    const p = await tx.payment.findUnique({ where: { id: paymentId }, select: { receiptCode: true, createdAt: true } });
    if (!p) return null;
    if (p.receiptCode) return p.receiptCode;
    const prefix = `R-${tashkentMonthKey(p.createdAt).slice(0, 4)}-`;
    const last = await tx.payment.findFirst({ where: { receiptCode: { startsWith: prefix } }, orderBy: { receiptCode: 'desc' }, select: { receiptCode: true } });
    const code = `${prefix}${nextSequence(last?.receiptCode ?? null, prefix)}`;
    await tx.payment.update({ where: { id: paymentId }, data: { receiptCode: code } });
    return code;
  }, TX));
}

export function allocateRefundReceiptCode(prisma: PrismaService, refundId: string): Promise<string | null> {
  return withRetry(() => prisma.$transaction(async (tx) => {
    const r = await tx.refund.findUnique({ where: { id: refundId }, select: { receiptCode: true } });
    if (!r) return null;
    if (r.receiptCode) return r.receiptCode;
    const prefix = `Q-${tashkentMonthKey(new Date()).slice(0, 4)}-`;
    const last = await tx.refund.findFirst({ where: { receiptCode: { startsWith: prefix } }, orderBy: { receiptCode: 'desc' }, select: { receiptCode: true } });
    const code = `${prefix}${nextSequence(last?.receiptCode ?? null, prefix)}`;
    await tx.refund.update({ where: { id: refundId }, data: { receiptCode: code } });
    return code;
  }, TX));
}
```
