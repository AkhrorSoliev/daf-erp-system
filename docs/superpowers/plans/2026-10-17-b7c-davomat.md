import { BadRequestException } from '@nestjs/common';
import { monthWindow, resolvePeriod } from './attendance-stats-period';

const TODAY = '2026-10-17';

describe('resolvePeriod', () => {
  it('a whole month compares with the month before and loads the six bars', () => {
    const p = resolvePeriod('2026-09-01', '2026-09-30', TODAY);
    expect(p.window).toEqual({ from: '2026-09-01', to: '2026-10-01' });
    expect(p.previous).toEqual({ startDate: '2026-08-01', endDate: '2026-08-31' });
    expect(p.previousWindow).toEqual({ from: '2026-08-01', to: '2026-09-01' });
    expect(p.months).toEqual(['2026-05', '2026-06', '2026-07', '2026-08', '2026-09']);
    expect(p.load).toEqual({ from: '2026-05-01', to: '2026-10-01' });
    expect(p.elapsedDays).toBe(30);
  });

  it('a custom range compares with the same number of days before it', () => {
    const p = resolvePeriod('2026-10-05', '2026-10-11', TODAY);
    expect(p.previous).toEqual({ startDate: '2026-09-28', endDate: '2026-10-04' });
    expect(p.elapsedDays).toBe(7);
  });

  it('counts the days only up to today (the current month runs to its last day)', () => {
    expect(resolvePeriod('2026-10-01', '2026-10-31', TODAY).elapsedDays).toBe(17);
    expect(resolvePeriod('2026-11-01', '2026-11-30', TODAY).elapsedDays).toBe(0);
  });

  it('a previous period before 2026-05-01 is null, not a false zero', () => {
    const may = resolvePeriod('2026-05-01', '2026-05-31', TODAY);
    expect(may.previous).toBeNull();
    expect(may.previousWindow).toBeNull();
    expect(may.months).toEqual(['2026-05']);
    expect(resolvePeriod('2026-06-01', '2026-08-31', TODAY).previous).toBeNull();
  });

  it('never shows more than six month bars', () => {
    const p = resolvePeriod('2027-01-01', '2027-03-31', '2027-04-10');
    expect(p.months).toEqual(['2026-10', '2026-11', '2026-12', '2027-01', '2027-02', '2027-03']);
    expect(p.load.from).toBe('2026-10-01');
  });

  it('refuses a start before 2026-05-01, in Uzbek', () => {
    expect(() => resolvePeriod('2026-04-30', '2026-05-31', TODAY)).toThrow(
      new BadRequestException('Bu hisobot 1-may 2026 dan boshlanadi'),
    );
  });

  it('refuses an unreal day and a reversed range (the 7a checks)', () => {
    expect(() => resolvePeriod('2026-02-30', '2026-03-01', TODAY)).toThrow(BadRequestException);
    expect(() => resolvePeriod('2026-09-30', '2026-09-01', TODAY)).toThrow(BadRequestException);
  });
});

  it('is the whole calendar month, upper bound exclusive', () => {
    expect(monthWindow('2026-12')).toEqual({ from: '2026-12-01', to: '2027-01-01' });
  });
});

  /** «Oylar bo'yicha»: up to six whole months ending at the period's last month, none before May 2026. */
  months: string[];
  /** Every day the one read must cover: the window, the previous window and the bars' whole months. */
  load: DayWindow;
  /** Days of the window up to today — what could have been marked (the min-row rule). */
  elapsedDays: number;
}

const dayWindow = (r: DayRange): DayWindow => ({
  from: r.startDate,
  to: addDaysToDateStr(r.endDate, 1),
});

/** A whole Tashkent calendar month as days, upper bound exclusive. */
export const monthWindow = (month: string): DayWindow => ({
  from: `${month}-01`,
  to: `${addMonthsToMonthKey(month, 1)}-01`,
});

const earliest = (...days: string[]) => days.reduce((a, b) => (b < a ? b : a));

/** `startDate`/`endDate` of the page → every window the report reads; a 400 for a bad period. */
export function resolvePeriod(
  startDate: string,
  endDate: string,
  today: string,
): StatsPeriod {
  // A real day, not reversed, at most 10 years: the 7a checks and texts.
  parseDayRange(startDate, endDate);
  if (startDate < ATTENDANCE_FLOOR) {
    throw new BadRequestException('Bu hisobot 1-may 2026 dan boshlanadi');
  }
  const range = { startDate, endDate };
  const window = dayWindow(range);
  const prev = previousRange(range);
  const previous = prev.startDate >= ATTENDANCE_FLOOR ? prev : null;
  const previousWindow = previous ? dayWindow(previous) : null;
  const months = monthKeysEndingAt(endDate, MONTHS_SHOWN).filter(
    (m) => m >= ATTENDANCE_FLOOR.slice(0, 7),
  );
  const load = {
    from: earliest(
      window.from,
      previousWindow?.from ?? window.from,
      monthWindow(months[0]).from,
    ),
    // The last bar's month ends on or after the window's end.
    to: monthWindow(months[months.length - 1]).to,
  };
  const until = endDate < today ? endDate : today;
  const elapsedDays =
    until < startDate
      ? 0
      : Math.round((Date.parse(until) - Date.parse(startDate)) / DAY_MS) + 1;
  return { range, window, previous, previousWindow, months, load, elapsedDays };
}

import type { AttendanceStatus } from '@prisma/client';
import { dayOfWeekForDateStr } from '../../common/date/tashkent';
import { round1 } from '../center-activity/center-metrics';
import type {
  AttRow,
  AttendanceFacts,
  Counts,
  CutKey,
  CutRow,
  DayWindow,
  GroupFact,
  MisserPair,
  WeekdayRow,
} from './attendance-stats.types';

/**
 * THE definition of the attendance rate (spec B7c §3, ADR-0103). The report,
 * its Excel file and the home page's «davomat (bu oy)» all count through this
 * file; no other place writes the formula.
 */

/** What the counting reads of a mark; `declaration: true` needs it exported (it is in the signatures). */
export type Mark = Pick<AttRow, 'date' | 'status' | 'cancelled'>;

const MISS_MIN = 3;

const emptyCounts = (): Counts => ({ came: 0, late: 0, absent: 0, excused: 0, cancelled: 0 });

/** A database mark → the report's mark: the `@db.Date` as its calendar day (= the Tashkent day). */
export function markOf(m: {
  date: Date;
  status: AttendanceStatus;
  cancellationId: string | null;
}): Mark {
  return {
    date: m.date.toISOString().slice(0, 10),
    status: m.status,
    cancelled: m.cancellationId !== null,
  };
}

/** A centre-cancelled lesson is its own count, never «Sababli». */
function addRow(c: Counts, r: Pick<AttRow, 'status' | 'cancelled'>): void {
  if (r.cancelled) {
    c.cancelled++;
    return;
  }
  if (r.status === 'PRESENT') c.came++;
  else if (r.status === 'LATE') c.late++;
  else if (r.status === 'ABSENT') c.absent++;
  else c.excused++;
}

export const inWindow = (date: string, w: DayWindow): boolean =>
  date >= w.from && date < w.to;

/** The five counts of the marks (inside `w` when given). */
export function countRows(rows: readonly Mark[], w?: DayWindow): Counts {
  const c = emptyCounts();
  for (const r of rows) if (!w || inWindow(r.date, w)) addRow(c, r);
  return c;
}

/** «Darslar»: the marks the rate is taken over (Keldi + Kechikdi + Kelmadi). */
export const lessonsOf = (c: Counts): number => c.came + c.late + c.absent;
/** Keldi + Kechikdi. */
export const attendedOf = (c: Counts): number => c.came + c.late;

/** (Keldi + Kechikdi) ÷ (Keldi + Kechikdi + Kelmadi), one decimal; null with nothing to divide by — never 0%. */
export function rateOf(c: Counts): number | null {
  const n = lessonsOf(c);
  return n > 0 ? round1((attendedOf(c) / n) * 100) : null;
}

/** Fewest marks a cut row needs to be listed (spec §4): ceil(60 × days / 30). */
export const minLessons = (days: number): number => Math.ceil((60 * days) / 30);

/** Monday–Saturday always, Sunday only when a mark falls on it; Σ = the window's counts. */
export function weekdayRows(rows: readonly Mark[], w: DayWindow): WeekdayRow[] {
  const byDay = new Map<number, Counts>([1, 2, 3, 4, 5, 6].map((d) => [d, emptyCounts()]));
  for (const r of rows) {
    if (!inWindow(r.date, w)) continue;
    const day = dayOfWeekForDateStr(r.date) || 7;
    let c = byDay.get(day);
    if (!c) {
      c = emptyCounts();
      byDay.set(day, c);
    }
    addRow(c, r);
  }
  return [...byDay.entries()]
    .sort(([a], [b]) => a - b)
    .map(([day, counts]) => ({ day, counts, rate: rateOf(counts) }));
}

/** The loader reads every group a mark points at (no deletedAt filter), so a miss is a broken load. */
export function groupOf(groups: ReadonlyMap<string, GroupFact>, id: string): GroupFact {
  const g = groups.get(id);
  if (!g) throw new Error(`Davomat: ${id} guruhi yuklanmagan`);
  return g;
}

/** The cut row a group belongs to: itself, its current teacher («Ustozsiz»: `none`), its current course. */
export function cutOf(
  g: GroupFact,
  cut: CutKey,
): { key: string; name: string; deleted: boolean } {
  if (cut === 'group') {
    return {
      key: g.id,
      name: g.teacherName ? `${g.name} · ${g.teacherName}` : g.name,
      deleted: g.deleted,
    };
  }
  if (cut === 'teacher') {
    return g.teacherId === null
      ? { key: 'none', name: 'Ustozsiz', deleted: false }
      : { key: String(g.teacherId), name: g.teacherName ?? '', deleted: false };
  }
  return { key: g.courseId, name: g.courseName, deleted: false };
}

/** Every (student, group) pair of the window with 3+ «Kelmadi» (spec §2, «Ko'p qoldiruvchi»). */
export function missPairs(rows: readonly AttRow[], w: DayWindow): MisserPair[] {
  const byPair = new Map<string, MisserPair>();
  for (const r of rows) {
    if (r.cancelled || r.status === 'EXCUSED' || !inWindow(r.date, w)) continue;
    const k = `${r.studentId}|${r.groupId}`;
    const p = byPair.get(k) ?? { studentId: r.studentId, groupId: r.groupId, absent: 0, lessons: 0 };
    p.lessons++;
    if (r.status === 'ABSENT') p.absent++;
    byPair.set(k, p);
  }
  return [...byPair.values()].filter((p) => p.absent >= MISS_MIN);
}

function lowestFirst(a: CutRow, b: CutRow): number {
  const ra = a.rate ?? Number.POSITIVE_INFINITY;
  const rb = b.rate ?? Number.POSITIVE_INFINITY;
  if (ra !== rb) return ra - rb;
  return b.lessons - a.lessons || a.name.localeCompare(b.name, 'uz', { numeric: true });
}

/**
 * Every row of a cut over the window — the min-row rule is the caller's — so
 * Σ rows = the window's counts (pinned by the overview parity test).
 */
export function cutRows(facts: AttendanceFacts, w: DayWindow, cut: CutKey): CutRow[] {
  const acc = new Map<string, { row: CutRow; misserIds: Set<number> }>();
  const entryOf = (groupId: string) => {
    const k = cutOf(groupOf(facts.groups, groupId), cut);
    let e = acc.get(k.key);
    if (!e) {
      e = { row: { ...k, counts: emptyCounts(), lessons: 0, rate: null, missers: 0 }, misserIds: new Set() };
      acc.set(k.key, e);
    }
    return e;
  };
  for (const r of facts.rows) if (inWindow(r.date, w)) addRow(entryOf(r.groupId).row.counts, r);
  for (const p of missPairs(facts.rows, w)) entryOf(p.groupId).misserIds.add(p.studentId);
  return [...acc.values()]
    .map(({ row, misserIds }) => ({
      ...row,
      lessons: lessonsOf(row.counts),
      rate: rateOf(row.counts),
      missers: misserIds.size,
    }))
    .sort(lowestFirst);
}

import { applyDiscount } from '../../billing/monthly-price';
import { groupOf, inWindow } from './attendance-counts';
import type {
  AttRow,
  ChargeFact,
  CreditResult,
  CreditSetting,
  DayWindow,
  GroupFact,
} from './attendance-stats.types';

const OFF: CreditSetting = { enabled: false, cap: null };
const keyOf = (studentId: number, groupId: string, month: string) =>
  `${studentId}|${groupId}|${month}`;

/** The «Sababli» marks the money block reads: in the window, not centre-cancelled, in a monthly course's group. */
export function excusedMonthly(
  rows: readonly AttRow[],
  w: DayWindow,
  groups: ReadonlyMap<string, GroupFact>,
): AttRow[] {
  return rows.filter(
    (r) =>
      r.status === 'EXCUSED' &&
      !r.cancelled &&
      inWindow(r.date, w) &&
      groupOf(groups, r.groupId).monthly,
  );
}

/**
 * «Keyingi oy to'lovidan taxminan X chegiriladi» (spec B7c §4.3). Per
 * (student, group, month): min(excused, the branch's monthly cap) × the
 * month's CHARGED per-lesson price after the frozen discount. An estimate: the
 * exact figure is the next month's `creditAmount`. The rule itself belongs to
 * billing (`MonthlyChargeService`); this only shows it.
 */
export function excusedCredit(
  excused: readonly AttRow[],
  groups: ReadonlyMap<string, GroupFact>,
  charges: readonly ChargeFact[],
  settings: ReadonlyMap<number, CreditSetting>,
): CreditResult {
  const chargeBy = new Map(
    charges.map((c) => [keyOf(c.studentId, c.groupId, c.month), c]),
  );
  const perKey = new Map<string, { n: number; branchId: number }>();
  for (const r of excused) {
    const k = keyOf(r.studentId, r.groupId, r.date.slice(0, 7));
    const e = perKey.get(k) ?? { n: 0, branchId: groupOf(groups, r.groupId).branchId };
    e.n++;
    perKey.set(k, e);
  }
  let amount = 0;
  let uncharged = 0;
  let anyOn = false;
  for (const [k, e] of perKey) {
    const s = settings.get(e.branchId) ?? OFF;
    if (s.enabled) anyOn = true;
    const charge = chargeBy.get(k);
    if (!charge) {
      uncharged += e.n;
      continue;
    }
    if (!s.enabled) continue;
    amount +=
      Math.min(e.n, s.cap ?? e.n) *
      applyDiscount(charge.perLessonCost, charge.discountPercent);
  }
  const enabled = perKey.size === 0 || anyOn;
  return {
    lessons: excused.length,
    students: new Set(excused.map((r) => r.studentId)).size,
    amount: enabled ? amount : null,
    enabled,
    uncharged,
  };
}

import { Injectable } from '@nestjs/common';
import { MonthlyChargeStatus, PaymentModel } from '@prisma/client';
import { utcMidnightFromDateStr } from '../../common/date/tashkent';
import {
  groupBranchWhere,
  type ReportBranchIds,
} from '../../common/finance/report-branch-scope';
import { PrismaService } from '../../prisma/prisma.service';
import { markOf } from './attendance-counts';
import type {
  AttendanceFacts,
  ChargeFact,
  DayWindow,
  GroupFact,
} from './attendance-stats.types';

export interface ChargeKey {
  studentId: number;
  groupId: string;
  month: string;
}

const fullName = (p: { firstName: string; lastName: string }) =>
  `${p.firstName} ${p.lastName}`.trim();
const monthKey = (y: number, m: number) => `${y}-${String(m).padStart(2, '0')}`;
const chargeKey = (k: ChargeKey) => `${k.studentId}|${k.groupId}|${k.month}`;

/**
 * Everything «Davomat» reads (spec B7c §6). One read of the marks per request.
 *
 * Deleted and closed groups are read on purpose (no `deletedAt` or status
 * filter): their lessons were held in that month (the 7b rule). The branch
 * comes through the group (`groupBranchWhere`). An empty scope never reaches
 * this class: the service refuses it first.
 */
@Injectable()
export class AttendanceFactsLoader {
  constructor(private readonly prisma: PrismaService) {}

  async load(
    companyId: number,
    scope: ReportBranchIds,
    w: DayWindow,
  ): Promise<AttendanceFacts> {
    const marks = await this.prisma.attendance.findMany({
      where: {
        companyId,
        date: {
          gte: utcMidnightFromDateStr(w.from),
          lt: utcMidnightFromDateStr(w.to),
        },
        ...groupBranchWhere(scope),
      },
      select: { groupId: true, studentId: true, date: true, status: true, cancellationId: true },
    });
    const rows = marks.map((m) => ({ groupId: m.groupId, studentId: m.studentId, ...markOf(m) }));
    const ids = [...new Set(rows.map((r) => r.groupId))];
    if (ids.length === 0) return { rows, groups: new Map() };

    const groups = await this.prisma.group.findMany({
      where: { id: { in: ids }, companyId },
      select: {
        id: true,
        name: true,
        branchId: true,
        deletedAt: true,
        courseId: true,
        course: { select: { name: true, paymentModel: true } },
        teachers: { select: { teacher: { select: { id: true, firstName: true, lastName: true } } } },
      },
    });
    return {
      rows,
      groups: new Map(
        groups.map((g): [string, GroupFact] => {
          // No group has two teachers today; if one ever does, the first by name stands for it (spec §2).
          const teacher = g.teachers
            .map((t) => ({ id: t.teacher.id, name: fullName(t.teacher) }))
            .sort((a, b) => a.name.localeCompare(b.name, 'uz'))[0];
          return [
            g.id,
            {
              id: g.id,
              name: g.name,
              branchId: g.branchId,
              deleted: g.deletedAt !== null,
              courseId: g.courseId,
              courseName: g.course.name,
              monthly: g.course.paymentModel === PaymentModel.MONTHLY,
              teacherId: teacher?.id ?? null,
              teacherName: teacher?.name ?? null,
            },
          ];
        }),
      ),
    };
  }

  /** The CHARGED month charges behind the «Sababli» marks (spec §4.3); exactly the asked keys. */
  async loadCharges(companyId: number, keys: readonly ChargeKey[]): Promise<ChargeFact[]> {
    if (keys.length === 0) return [];
    const months = [...new Set(keys.map((k) => k.month))];
    const rows = await this.prisma.enrollmentMonthlyCharge.findMany({
      where: {
        companyId,
        status: MonthlyChargeStatus.CHARGED,
        studentId: { in: [...new Set(keys.map((k) => k.studentId))] },
        groupId: { in: [...new Set(keys.map((k) => k.groupId))] },
        OR: months.map((m) => ({ periodYear: Number(m.slice(0, 4)), periodMonth: Number(m.slice(5, 7)) })),
      },
      select: {
        studentId: true,
        groupId: true,
        periodYear: true,
        periodMonth: true,
        perLessonCost: true,
        discountPercent: true,
      },
    });
    const wanted = new Set(keys.map(chargeKey));
    return rows
      .map((c) => ({
        studentId: c.studentId,
        groupId: c.groupId,
        month: monthKey(c.periodYear, c.periodMonth),
        perLessonCost: c.perLessonCost,
        discountPercent: c.discountPercent,
      }))
      .filter((c) => wanted.has(chargeKey(c)));
  }

  /** «Ism Familiya» of the «ko'p qoldiruvchi» list; an archived card keeps its name. */
  async loadStudentNames(companyId: number, ids: readonly number[]): Promise<Map<number, string>> {
    const unique = [...new Set(ids)];
    if (unique.length === 0) return new Map();
    const students = await this.prisma.student.findMany({
      where: { id: { in: unique }, companyId },
      select: { id: true, firstName: true, lastName: true },
    });
    return new Map(students.map((s) => [s.id, fullName(s)]));
  }
}

import {
  attendedOf,
  countRows,
  cutOf,
  cutRows,
  groupOf,
  lessonsOf,
  minLessons,
  missPairs,
  rateOf,
  weekdayRows,
} from './attendance-counts';
import { monthWindow, type StatsPeriod } from './attendance-stats-period';
import type {
  AttendanceFacts,
  CreditResult,
  CutKey,
  DayWindow,
  MisserPair,
  OverviewCore,
} from './attendance-stats.types';

export interface OverviewInput {
  facts: AttendanceFacts;
  period: StatsPeriod;
  /** The current Tashkent month: its bar (and any later one) is «oy tugamagan». */
  currentMonth: string;
  credit: CreditResult;
  cut: CutKey;
  page: number;
  pageSize: number;
}

/**
 * The whole page from one mark set (spec B7c §4). Card, month bars, weekdays
 * and the three cuts all count through `countRows`, so they add up (pinned by
 * the parity test). The cut is paged here; a page past the end is clamped.
 */
export function buildOverview(i: OverviewInput): OverviewCore {
  const { facts, period } = i;
  const counts = countRows(facts.rows, period.window);
  const all = cutRows(facts, period.window, i.cut);
  const min = minLessons(period.elapsedDays);
  const listed = all.filter((r) => r.lessons >= min);
  const page = Math.min(i.page, Math.max(1, Math.ceil(listed.length / i.pageSize)));
  return {
    range: period.range,
    previous: period.previous,
    card: {
      counts,
      lessons: lessonsOf(counts),
      attended: attendedOf(counts),
      rate: rateOf(counts),
      previousRate: period.previousWindow
        ? rateOf(countRows(facts.rows, period.previousWindow))
        : null,
    },
    months: period.months.map((month) => {
      const c = countRows(facts.rows, monthWindow(month));
      return { month, counts: c, rate: rateOf(c), partial: month >= i.currentMonth };
    }),
    weekdays: weekdayRows(facts.rows, period.window),
    credit: i.credit,
    cut: {
      key: i.cut,
      rows: listed.slice((page - 1) * i.pageSize, page * i.pageSize),
      total: listed.length,
      hidden: all.length - listed.length,
      minLessons: min,
      page,
      pageSize: i.pageSize,
    },
  };
}

/** The «ko'p qoldiruvchi» pairs of one cut row (the dialog's list, before names). */
export function missersOf(
  facts: AttendanceFacts,
  w: DayWindow,
  cut: CutKey,
  key: string,
): MisserPair[] {
  return missPairs(facts.rows, w).filter(
    (p) => cutOf(groupOf(facts.groups, p.groupId), cut).key === key,
  );
}

import { ForbiddenException, Injectable } from '@nestjs/common';
import { tashkentDateStr, tashkentMonthKey } from '../../common/date/tashkent';
import {
  isEmptyScope,
  type ReportBranchIds,
} from '../../common/finance/report-branch-scope';
import { AbsenceStreakService } from '../../outreach/absence-streak.service';
import { SettingsService } from '../../settings/settings.service';
import { groupOf } from './attendance-counts';
import { excusedCredit, excusedMonthly } from './attendance-credit';
import { AttendanceFactsLoader } from './attendance-facts.loader';
import { buildOverview, missersOf } from './attendance-overview';
import { resolvePeriod, type StatsPeriod } from './attendance-stats-period';
import type {
  AttendanceFacts,
  AttendanceOverview,
  CreditResult,
  CreditSetting,
  MisserPair,
  MisserRow,
  MissersPage,
} from './attendance-stats.types';
import type {
  AttendanceMissersQueryDto,
  AttendanceStatsQueryDto,
} from './dto/attendance-stats-query.dto';

/** An empty scope is refused, never served as zeros (ADR-0002). */
function assertScope(scope: ReportBranchIds) {
  if (isEmptyScope(scope)) {
    throw new ForbiddenException("Bu filial ma'lumotlarini ko'rish huquqingiz yo'q");
  }
}

const pageOf = (total: number, page: number, pageSize: number) =>
  Math.min(page, Math.max(1, Math.ceil(total / pageSize)));

/**
 * «Davomat» (spec B7c, ADR-0103). One read of the marks, then pure functions.
 *
 * // ponytail: no cache — six months are ~35k marks and every cut/page click
 * reads them again; add Redis (key: company + scope + startDate + endDate) if
 * it is measured slow.
 */
@Injectable()
export class AttendanceStatsService {
  constructor(
    private readonly loader: AttendanceFactsLoader,
    private readonly settings: SettingsService,
    private readonly streaks: AbsenceStreakService,
  ) {}

  /** `GET /reports/attendance-stats/overview`. */
  async overview(
    companyId: number,
    scope: ReportBranchIds,
    q: AttendanceStatsQueryDto,
  ): Promise<AttendanceOverview> {
    assertScope(scope);
    const now = new Date();
    const period = resolvePeriod(q.startDate, q.endDate, tashkentDateStr(now));
    const [facts, streakRows] = await Promise.all([
      this.loader.load(companyId, scope, period.load),
      // «Ketma-ket kelmayotganlar» is today's state, not the period's (spec §4).
      this.streaks.computeStreaks({ companyId, branchIds: scope ?? undefined, threshold: 3 }),
    ]);
    const credit = await this.credit(companyId, facts, period);
    return {
      ...buildOverview({
        facts,
        period,
        currentMonth: tashkentMonthKey(now),
        credit,
        cut: q.cut ?? 'group',
        page: q.page ?? 1,
        pageSize: q.pageSize ?? 10,
      }),
      streak: new Set(streakRows.map((r) => r.studentId)).size,
    };
  }

  /** `GET /reports/attendance-stats/missers`: the «ko'p qoldiruvchi» list of one cut row, paged. */
  async missers(
    companyId: number,
    scope: ReportBranchIds,
    q: AttendanceMissersQueryDto,
  ): Promise<MissersPage> {
    assertScope(scope);
    const period = resolvePeriod(q.startDate, q.endDate, tashkentDateStr(new Date()));
    const facts = await this.loader.load(companyId, scope, period.window);
    const rows = await this.named(
      companyId,
      facts,
      missersOf(facts, period.window, q.cut ?? 'group', q.key),
    );
    const pageSize = q.pageSize ?? 10;
    const page = pageOf(rows.length, q.page ?? 1, pageSize);
    return {
      rows: rows.slice((page - 1) * pageSize, page * pageSize),
      total: rows.length,
      page,
      pageSize,
    };
  }

  /** The «Sababli» money block: charges and the per-branch setting behind its marks. */
  private async credit(
    companyId: number,
    facts: AttendanceFacts,
    period: StatsPeriod,
  ): Promise<CreditResult> {
    const excused = excusedMonthly(facts.rows, period.window, facts.groups);
    const branchIds = [...new Set(excused.map((r) => groupOf(facts.groups, r.groupId).branchId))];
    const [charges, settings] = await Promise.all([
      this.loader.loadCharges(
        companyId,
        excused.map((r) => ({ studentId: r.studentId, groupId: r.groupId, month: r.date.slice(0, 7) })),
      ),
      Promise.all(
        branchIds.map(async (b): Promise<[number, CreditSetting]> => [
          b,
          {
            enabled: await this.settings.get(companyId, 'payment.excusedCreditEnabled', b),
            cap: await this.settings.get(companyId, 'payment.excusedCreditMonthlyCap', b),
          },
        ]),
      ),
    ]);
    return excusedCredit(excused, facts.groups, charges, new Map(settings));
  }

  /** Names the pairs; most absences first, then the name (uz, numeric). */
  private async named(
    companyId: number,
    facts: AttendanceFacts,
    pairs: MisserPair[],
  ): Promise<MisserRow[]> {
    const names = await this.loader.loadStudentNames(companyId, pairs.map((p) => p.studentId));
    return pairs
      .map((p) => ({
        ...p,
        name: names.get(p.studentId) ?? `#${p.studentId}`,
        groupName: groupOf(facts.groups, p.groupId).name,
      }))
      .sort(
        (a, b) =>
          b.absent - a.absent ||
          a.name.localeCompare(b.name, 'uz', { numeric: true }) ||
          a.groupName.localeCompare(b.groupName, 'uz', { numeric: true }),
      );
  }
}

import { Workbook, type Worksheet } from 'exceljs';
import type {
  CutKey,
  CutRow,
  MisserRow,
  OverviewCore,
} from './attendance-stats.types';

export interface AttendanceExcelInput {
  overview: OverviewCore;
  /** Every row of every cut — the min-row rule does not apply in the file (spec §9). */
  cuts: Record<CutKey, CutRow[]>;
  missers: MisserRow[];
}

const CUT_NAME: Record<CutKey, string> = { group: 'Guruh', teacher: 'Ustoz', course: 'Kurs' };
const CUT_ORDER: CutKey[] = ['group', 'teacher', 'course'];
const days = (r: { startDate: string; endDate: string }) => `${r.startDate} — ${r.endDate}`;

function header(sheet: Worksheet, titles: [string, number][]) {
  sheet.columns = titles.map(([title, width]) => ({ header: title, width }));
  sheet.getRow(1).font = { bold: true };
}

/** The workbook of the page (spec B7c §9): the same `buildOverview` answer, every cut row, every 3+ pair. */
export async function attendanceWorkbook(i: AttendanceExcelInput): Promise<Buffer> {
  const wb = new Workbook();
  const o = i.overview;
  const c = o.card.counts;

  const sum = wb.addWorksheet('Umumiy');
  header(sum, [["Ko'rsatkich", 40], ['Qiymat', 22]]);
  const lines: [string, number | string][] = [
    ['Davr', days(o.range)],
    ['Davomat, %', o.card.rate ?? ''],
    ['Oldingi davr', o.previous ? days(o.previous) : ''],
    ['Oldingi davr davomati, %', o.card.previousRate ?? ''],
    ["O'tilishi kerak bo'lgan darslar", o.card.lessons],
    ['Kelishgan', o.card.attended],
    ['Keldi', c.came],
    ['Kechikdi', c.late],
    ['Kelmadi', c.absent],
    ['Sababli (foizga kirmaydi)', c.excused],
    ['Markaz bekor qildi (foizga kirmaydi)', c.cancelled],
    ['Sababli darslar (oylik kurslar)', o.credit.lessons],
    ["Sababli darsli o'quvchilar", o.credit.students],
    ["Taxminiy chegirma, so'm", o.credit.amount ?? "chegirma o'chirilgan"],
  ];
  for (const line of lines) sum.addRow(line);

  const months = wb.addWorksheet('Oylar');
  header(months, [['Oy', 12], ['Davomat, %', 12], ['Keldi', 10], ['Kechikdi', 10], ['Kelmadi', 10], ['Sababli', 10], ['Markaz bekor qildi', 18], ['Izoh', 16]]);
  for (const m of o.months) {
    months.addRow([
      m.month, m.rate ?? '', m.counts.came, m.counts.late, m.counts.absent,
      m.counts.excused, m.counts.cancelled, m.partial ? 'oy tugamagan' : '',
    ]);
  }

  const cut = wb.addWorksheet('Kesim');
  header(cut, [['Kesim', 10], ['Nomi', 34], ['Davomat, %', 12], ['Darslar', 10], ['Keldi', 10], ['Kechikdi', 10], ['Kelmadi', 10], ['Sababli', 10], ["Ko'p qoldiruvchi", 16], ['Izoh', 14]]);
  for (const k of CUT_ORDER) {
    for (const r of i.cuts[k]) {
      cut.addRow([
        CUT_NAME[k], r.name, r.rate ?? '', r.lessons, r.counts.came, r.counts.late,
        r.counts.absent, r.counts.excused, r.missers, r.deleted ? "o'chirilgan" : '',
      ]);
    }
  }

  const students = wb.addWorksheet("O'quvchilar");
  header(students, [["O'quvchi", 28], ['Guruh', 22], ['Kelmadi', 10], ['Jami', 10]]);
  for (const m of i.missers) students.addRow([m.name, m.groupName, m.absent, m.lessons]);

  return Buffer.from(await wb.xlsx.writeBuffer());
}

import { resolveDepartedRange } from "@/components/reports/departed-students/departed-url";
import type { CutKey } from "./attendance-types";

/** The report starts with the reporting base (spec B7c §3); the server answers 400 before it. */
export const ATT_FLOOR = "2026-05-01";
export const ATT_PAGE_SIZES = [10, 20, 30, 40, 50];

/** URL keys; defaults are left out of the URL by `useUrlFilters`. The old `att_*` keys are not read. */
export const ATTENDANCE_SCHEMA = {
  preset: { type: "string", defaultValue: "current_month" },
  startDate: { type: "string", defaultValue: "" },
  endDate: { type: "string", defaultValue: "" },
  cut: { type: "string", defaultValue: "group" },
  page: { type: "number", defaultValue: 1 },
  pageSize: { type: "number", defaultValue: 10 },
} as const;

export interface AttendanceFilters {
  preset: string;
  startDate: string;
  endDate: string;
  cut: string;
  page: number;
  pageSize: number;
}

/** «Qayta urinish» on an error with a non-default URL goes back here. */
export const ATTENDANCE_DEFAULTS: AttendanceFilters = {
  preset: "current_month", startDate: "", endDate: "", cut: "group", page: 1, pageSize: 10,
};

const CUTS: readonly string[] = ["group", "teacher", "course"];

/** An unknown `?cut=` is the group cut. */
export const cleanCut = (v: string): CutKey => (CUTS.includes(v) ? (v as CutKey) : "group");

export interface DayRange {
  startDate: string;
  endDate: string;
}

/**
 * The days a preset stands for (7a's buttons), never before 2026-05-01: a start
 * before it is lifted to it, a period wholly before it becomes this month.
 * `today` is the Tashkent day as a local date.
 */
export function attendanceRange(f: Pick<AttendanceFilters, "preset" | "startDate" | "endDate">, today: Date): DayRange {
  const r = resolveDepartedRange(f.preset, f.startDate, f.endDate, today);
  if (r.endDate < ATT_FLOOR) return resolveDepartedRange("current_month", "", "", today);
  return r.startDate < ATT_FLOOR ? { startDate: ATT_FLOOR, endDate: r.endDate } : r;
}

const cleanPage = (n: number): number => (Number.isInteger(n) && n >= 1 ? n : 1);
const cleanSize = (n: number): number => (ATT_PAGE_SIZES.includes(n) ? n : 10);

export interface OverviewParams extends DayRange {
  cut: CutKey;
  page: number;
  pageSize: number;
}

/** The filters as `GET /reports/attendance-stats/overview` reads them. */
export function overviewParams(f: AttendanceFilters, today: Date): OverviewParams {
  return { ...attendanceRange(f, today), cut: cleanCut(f.cut), page: cleanPage(f.page), pageSize: cleanSize(f.pageSize) };
}

export interface MissersParams extends OverviewParams {
  key: string;
}

/** One cut row's «ko'p qoldiruvchi» page, for `GET /reports/attendance-stats/missers`. */
export function missersParams(range: DayRange, cut: CutKey, key: string, page: number, pageSize: number): MissersParams {
  return { ...range, cut, key, page: cleanPage(page), pageSize: cleanSize(pageSize) };
}

export const excelFileName = (r: DayRange): string => `davomat-${r.startDate}_${r.endDate}.xlsx`;

/** «avgustda 79% edi»; null when there is no earlier figure (before 2026-05 or nothing to divide by). */
export function previousText(preset: string, previous: { startDate: string } | null, previousRate: number | null): string | null {
  if (!previous || previousRate === null) return null;
  return `${previousLabel(preset, previous)} ${pctText(previousRate)} edi`;
}

export const lessonsText = (lessons: number, attended: number): string =>
  `o'tilishi kerak bo'lgan ${formatNumber(lessons)} darsdan ${formatNumber(attended)} tasiga kelishgan`;

/** The money block's sentence (spec §4.3); every number is the server's. */
export function creditText(c: CreditResult): string {
  if (c.lessons === 0) return "Bu davrda oylik kurslarda sababli dars yo'q.";
  const head = `${formatNumber(c.lessons)} ta sababli dars (${formatNumber(c.students)} o'quvchi)`;
  if (!c.enabled || c.amount === null) return `${head}. Sababli dars chegirmasi o'chirilgan.`;
  return `${head} uchun keyingi oy to'lovidan taxminan ${mlnText(c.amount)} chegiriladi.`;
}

export const unchargedText = (n: number): string | null =>
  n > 0 ? `Shundan ${formatNumber(n)} tasining oyi hali hisoblanmagan, summaga kirmadi.` : null;

export const hiddenText = (hidden: number, min: number): string | null =>
  hidden > 0 ? `${formatNumber(hidden)} ta qatorda yozuv kam (${formatNumber(min)} tadan kam), ro'yxatda yo'q.` : null;

export const streakText = (n: number): string => `${formatNumber(n)} o'quvchi 3 va undan ko'p darsga ketma-ket sababsiz kelmadi.`;

# ADR-0103: Davomat hisoboti — hamma blok bitta yozuvlar to'plamidan, foiz bitta funksiyadan

**Holati:** Qabul qilindi
**Sana:** 2026-10-17
**Bog'liq:** spec `docs/superpowers/specs/2026-10-17-b7c-davomat-design.md`, ADR-0098 (7a, bitta ma'lumot to'plami), ADR-0099 (7b), ADR-0054 («Dars bo'ldimi?»), ADR-0053 (bekor qilingan dars puli), ADR-0067 (hisobotlar bazasi 2026-05), ADR-0002 (filial qamrovi)

## Kontekst

Eski `/reports/attendance` (`reports-attendance-analytics.service.ts`, 1078 qator) ishonchsiz edi:

- Umumiy foiz jadvallar yig'indisiga mos kelmasdi: umumiy hisob hamma guruhni, guruhlar jadvali faol/shakllanayotgan/tugagan guruhlarni, kurslar jadvali hamma guruhni, ustozlar jadvali faqat faol va shakllanayotganlarni olardi.
- Markaz bekor qilgan darslar (`EXCUSED` + `cancellationId`) «Sababli» ichiga aralashardi.
- Bosh sahifadagi davomat boshqa kodda hisoblanardi (`getKpis`) va hisobotdan farq qilishi mumkin edi.
- Bo'sh davr 0% bo'lib chizilardi; 1–2 o'quvchili guruhning bitta kelmasligi 50% berib, ro'yxat boshiga chiqardi.

## Qaror

1. **Manba — `Attendance` yozuvlari**, bir yozuv = bir o'quvchi, bir guruh, bir kun. `AttendanceFactsLoader` davr, oldingi davr va oylar ustunining eng erta kunidan oxirigacha yozuvlarni BIR marta o'qiydi (`groupBranchWhere` qamrovi bilan); o'chirilgan va yopilgan guruhlar ATAYLAB o'qiladi.
2. **Holatlar:** Keldi = `PRESENT`, Kechikdi = `LATE`, Kelmadi = `ABSENT`, Sababli = `EXCUSED` va `cancellationId` yo'q, Markaz bekor qildi = `cancellationId` bor. Oxirgi ikkitasi hech bir foizga kirmaydi.
3. **Foiz** = (Keldi + Kechikdi) ÷ (Keldi + Kechikdi + Kelmadi), bir kasr (`round1`); maxraj 0 bo'lsa `null` («—»). Yagona ta'rif — `reports/attendance-stats/attendance-counts.ts` (`countRows`, `rateOf`); karta, oylar, hafta kunlari, uch kesim, Excel va bosh sahifa shu funksiyadan.
4. **Davr:** 7a'ning tugmalari (`startDate`/`endDate`), eng erta kun 2026-05-01 (oldingisi 400). Oldingi davr: to'liq oy → oldingi oy, aks holda shu uzunlikdagi oldingi davr; 2026-05-01 dan oldin boshlansa `null`. Oylar ustuni: maydan davr oxirgi oyigacha, ko'pi bilan 6 ta, har biri to'liq kalendar oyi. Hafta kuni — yozuv sanasining kalendar kuni; yakshanba faqat yozuvi bo'lsa.
5. **Kesimlar:** guruh («A1-04 · Ustoz», o'chirilgan belgisi bilan), ustoz (guruhning hozirgi ustozi; bir nechta bo'lsa ism bo'yicha birinchisi; ustozsiz guruhlar «Ustozsiz»), kurs (guruhning hozirgi kursi). Qator ro'yxatda chiqishi uchun `ceil(60 × kun / 30)` yozuv kerak (kun bugungacha sanaladi); yashirin qatorlar umumiy sonlarga baribir kiradi — kesimlar yig'indisi umumiyga teng (parite testi). Kesim server tomonda sahifalanadi (10–50).
6. **Ko'p qoldiruvchi:** davrda bir guruhda 3+ `ABSENT` yozuvi bo'lgan (o'quvchi, guruh) juftliklari; qatordagi son — alohida o'quvchilar; ro'yxat `GET /reports/attendance-stats/missers` dan (o'quvchi, guruh, kelmadi, jami = Keldi + Kechikdi + Kelmadi).
7. **Sababli darslar puli (taxmin):** oylik kurs guruhlaridagi Sababli yozuvlar; har (o'quvchi, guruh, oy) uchun o'sha oyning `CHARGED` hisobidan `min(soni, payment.excusedCreditMonthlyCap) × applyDiscount(perLessonCost, discountPercent)`; sozlama guruh filialiniki; `payment.excusedCreditEnabled` o'chiq filial summaga kirmaydi; hisobi yo'q oy faqat sonni beradi.

9. **Bosh sahifa:** `getKpis().averageAttendance` = `rateOf(countRows(…))` joriy Toshkent oyi 1-kunidan bugungacha, tanlangan filial; bir kasr yoki `null`; sahifa butun songa yaxlitlaydi.
10. **Yo'llar:** `GET /reports/attendance-stats/overview|missers|excel`, `reports.students`, `@BranchScope()` ro'yxat sifatida, bo'sh qamrov 403. Eski `attendance-analytics`, `attendance-by-group`, `attendance-by-course`, `teacher-performance` olib tashlandi; `getAttendanceAnalytics` (bosh sahifa haftalik grafigi, moliyaviy Excel) va `getTeacherPerformance` (moliyaviy Excel) qoldi. Kesh yo'q.

## Ko'rib chiqilgan muqobillar

- **Eski servisni tuzatish** — to'rt xil guruh to'plami bitta faylda; har tuzatish boshqa jadvalni buzardi. Rad.
- **Kesim va sahifalashni mijozda** — mijoz hech narsa hisoblamasin (spec §7); kesim har bosishda serverdan keladi.
- **Ustozni dars kunidagi o'rinbosar bo'yicha** — CEO qarori: guruhning ustozi; hozir ikki ustozli guruh yo'q.

## Oqibatlari

- Kartadagi raqam hafta kunlari, oylar ustuni va har kesim yig'indisiga teng.
- Bosh sahifadagi davomat endi «0%» emas, yozuv bo'lmasa «—»; qiymat hisobotdagi «Bu oy» raqami bilan bir xil.

## Davomat foizi qanday hisoblanadi

- Foiz = (Keldi + Kechikdi) ÷ (Keldi + Kechikdi + Kelmadi). Kechikib kelgan o'quvchi kelgan hisoblanadi.

- **Davomat kartasi** — foiz, oldingi davrdagi foiz, «o'tilishi kerak bo'lgan N darsdan M tasiga kelishgan» va to'rtta belgi: Keldi, Kechikdi, Kelmadi, Sababli. Markaz bekor qilgan darslar bo'lsa, ular soni kichik qatorda.
